newcontest() {
    base="$HOME/Projects/CP/contest/contest-$(date +%Y-%m-%d)"
    folder="$base"

    if [ -d "$folder" ]; then
        i=1
        while [ -d "$base ($i)" ]; do
            ((i++))
        done
        folder="$base ($i)"
    fi

    mkdir -p "$folder"
    cd "$folder"
    code .
}
→ newcontest = New dated contest folder + VS Code
